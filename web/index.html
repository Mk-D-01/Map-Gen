perlin_noise.py 
import math

def hash_coords(x: int, y: int, seed: int) -> float:
    n = (x + y * 57 + seed * 131) & 0xFFFFFFFF
    n = ((n << 13) ^ n) & 0xFFFFFFFF
    term = (n * (n * n * 15731 + 789221) + 1376312589) & 0x7FFFFFFF
    return 1.0 - (term / 1073741824.0)

def smooth_fade(t: float) -> float:
    # Quintic Hermite polynomial: 6t^5 - 15t^4 + 10t^3
    return t * t * t * (t * (t * 6.0 - 15.0) + 10.0)

def smooth_noise_2d(x: float, y: float, seed: int) -> float:
    X, Y = math.floor(x), math.floor(y)
    xf, yf = x - X, y - Y
    u, v = smooth_fade(xf), smooth_fade(yf)

    n00 = (hash_coords(X, Y, seed) + 1.0) * 0.5
    n10 = (hash_coords(X + 1, Y, seed) + 1.0) * 0.5
    n01 = (hash_coords(X, Y + 1, seed) + 1.0) * 0.5
    n11 = (hash_coords(X + 1, Y + 1, seed) + 1.0) * 0.5

    x1 = n00 + u * (n10 - n00)
    x2 = n01 + u * (n11 - n01)
    return x1 + v * (x2 - x1)
cellular_cave.py 
def generate_cellular_cave(width: int, height: int, seed: int, fill_prob=0.45, passes=5):
    # Step 1: Initialize random wall grid with solid boundary
    grid = [[1 if x == 0 or x == width-1 or y == 0 or y == height-1 else
             (1 if (hash_coords(x, y, seed) + 1) * 0.5 < fill_prob else 0)
             for x in range(width)] for y in range(height)]

    # Step 2: Apply 4-5 Rule with State Hysteresis
    for _ in range(passes):
        new_grid = [[1] * width for _ in range(height)]
        for y in range(1, height - 1):
            for x in range(1, width - 1):
                walls = sum(grid[y+dy][x+dx] for dy in (-1,0,1) for dx in (-1,0,1) if not (dx==0 and dy==0))
                if walls > 4: new_grid[y][x] = 1
                elif walls < 4: new_grid[y][x] = 0
                else: new_grid[y][x] = grid[y][x] # Hysteresis retain
        grid = new_grid
    return grid
bsp_dungeon.py 
class BSPNode:
    def __init__(self, x, y, w, h):
        self.x, self.y, self.w, self.h = x, y, w, h
        self.left = None
        self.right = None
        self.room = None

    def split(self, min_size: int, seed: int) -> bool:
        if self.left or self.right: return False
        split_h = (self.h / self.w >= 1.25) if (self.w / self.h < 1.25) else False
        max_size = (self.h if split_h else self.w) - min_size
        if max_size <= min_size: return False
        
        split_pos = min_size + int((max_size - min_size) * 0.5)
        if split_h:
            self.left = BSPNode(self.x, self.y, self.w, split_pos)
            self.right = BSPNode(self.x, self.y + split_pos, self.w, self.h - split_pos)
        else:
            self.left = BSPNode(self.x, self.y, split_pos, self.h)
            self.right = BSPNode(self.x + split_pos, self.y, self.w - split_pos, self.h)
        return True
